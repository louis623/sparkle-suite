const {chromium}=require(process.env.PLAYWRIGHT_PACKAGE || '/usr/local/lib/pnpm/5/.pnpm/playwright@1.63.0-alpha-2026-08-31/node_modules/playwright');const fs=require('fs');
const themes=['amethyst','gilded_autumn','halloween_pumpkin_cat','midnight_rose','fall_owl_harvest_moon','gobble_and_fizz','pie_please','pearl_rose','sunset_beach','rose_champagne','alpine_opal','gnome_garden','neon_butterfly','halloween_pumpkin_witch'];
(async()=>{const b=await chromium.launch({executablePath:'/opt/google/chrome/chrome',headless:true,args:['--no-sandbox']}),rows=[];
for(const theme of themes)for(const device of ['desktop','phone'])for(const reducedMotion of ['no-preference','reduce']){
const c=await b.newContext({viewport:device==='phone'?{width:390,height:844}:{width:1440,height:900},reducedMotion});await c.route('**/*',r=>new URL(r.request().url()).hostname==='127.0.0.1'?r.continue():r.abort());const p=await c.newPage();await p.goto(`http://127.0.0.1:4398/review?theme=${theme}&page=homepage&renderer=ssr`);let state;
if(await p.locator('video').count()){
 if(reducedMotion==='no-preference'){
  await p.waitForFunction(()=>{const v=document.querySelector('video');return !v.paused&&v.currentTime>.06&&Number.isFinite(v.duration)},{timeout:15000});
  const start=await p.locator('video').evaluate(v=>({time:v.currentTime,duration:v.duration,loop:v.loop,muted:v.muted}));await p.waitForTimeout(200);const advances=await p.locator('video').evaluate((v,start)=>v.currentTime>start,start.time);
  await p.locator('video').evaluate(v=>new Promise(resolve=>{v.addEventListener('seeked',resolve,{once:true});v.currentTime=v.duration-.25}));await p.waitForTimeout(700);
  state=await p.locator('video').evaluate((v,x)=>({...x,afterLoop:v.currentTime,loopPassed:v.loop&&!v.paused&&v.currentTime<2,paused:v.paused}),{...start,advances});
 }else{await p.waitForTimeout(350);state=await p.locator('video').evaluate(v=>({paused:v.paused,time:v.currentTime,still:document.querySelector('[data-hero-still]')?.getAttribute('src'),src:v.getAttribute('src')}))}
}else{
 await p.waitForTimeout(100);const first=await p.evaluate(()=>document.getAnimations().filter(a=>a.effect.target?.closest('[data-ssr-hero],[data-hero-fx]')).map(a=>({name:a.animationName,time:a.currentTime,state:a.playState,iterations:a.effect.getTiming().iterations})));await p.waitForTimeout(150);state=await p.evaluate(first=>{const animations=document.getAnimations().filter(a=>a.effect.target?.closest('[data-ssr-hero],[data-hero-fx]'));return {animations:first,advances:animations.some((a,i)=>a.playState==='running'&&a.currentTime>first[i]?.time),loops:animations.filter(a=>a.playState==='running'&&a.effect.getTiming().iterations!==1).every(a=>a.effect.getTiming().iterations===Infinity),reducedStill:animations.every(a=>a.playState!=='running')}} ,first);
}
rows.push({theme,device,reducedMotion,...state});fs.writeFileSync('.tmp-qa/motion.json',JSON.stringify(rows,null,2));console.log(theme,device,reducedMotion,JSON.stringify(state));await c.close()}
await b.close()})().catch(e=>{console.error(e);process.exit(1)});
