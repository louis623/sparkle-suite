const {chromium}=require('./node_modules/playwright'),fs=require('node:fs');
(async()=>{
const browser=await chromium.launch({channel:'chrome',headless:true}),results=[];
try{for(const theme of ['alpine_opal','gilded_autumn']){
 const context=await browser.newContext({viewport:{width:390,height:844},reducedMotion:'no-preference'}),writes=[],errors=[];
 await context.route('**/*',route=>{
  const request=route.request(),url=new URL(request.url());
  if(request.method()!=='GET'){writes.push(request.method());return route.abort()}
  if(url.hostname!=='127.0.0.1')return route.abort();
  const page={'/':'homepage','/trade':'trade','/join':'join'}[url.pathname];
  if(page)return route.fulfill({status:200,contentType:'text/html',body:fs.readFileSync(`.tmp-qa/html/${theme}-${page}-ssr.html`)});
  if(url.pathname==='/api/amethyst/trade-board')return route.fulfill({status:200,contentType:'application/json',body:JSON.stringify({listings:[{id:'fixture-only',name:'Fixture Aurora Earrings',collection:'Originals',type:'Earrings',material:'Silver',stone:'CZ',size:null,note:'Local fixture',glyph:'S',tier:'everyday',photoUrl:null,photoSource:'missing'}]})});
  return route.continue();
 });
 const page=await context.newPage();page.on('pageerror',e=>errors.push(e.message));
 await page.goto(`http://127.0.0.1:4397/review?theme=${theme}&page=homepage&renderer=ssr`,{waitUntil:'load'});
 const states=[];
 for(const target of ['/trade','/join','/']){
  await page.locator(`a[href="${target}"]`).first().click();await page.locator('h1').waitFor();
  const state=await page.evaluate(()=>({path:location.pathname,h1:document.querySelector('h1')?.textContent,serverRendered:document.getElementById('root')?.dataset.serverRendered}));
  if(state.path!==target||!state.h1||state.serverRendered!=='true')throw Error('Local navigation mismatch');states.push(state);
 }
 let controls;
 if(theme==='alpine_opal'){
  await page.emulateMedia({reducedMotion:'reduce'});await page.waitForFunction(()=>document.querySelector('video')?.paused===true);
  const paused=await page.locator('video').evaluate(v=>v.paused);
  await page.emulateMedia({reducedMotion:'no-preference'});await page.waitForFunction(()=>document.querySelector('video')?.paused===false);
  controls={reducedMotionChangePaused:paused,normalMotionResumed:await page.locator('video').evaluate(v=>!v.paused)};
 }else{
  await page.getByRole('button',{name:'Pause animation',exact:true}).click();await page.waitForFunction(()=>document.querySelector('video')?.paused===true);
  await page.getByRole('button',{name:'Play animation',exact:true}).click();await page.waitForFunction(()=>document.querySelector('video')?.paused===false);
  controls={pauseAndPlay:true};
 }
 results.push({theme,states,controls,errors,writes});await context.close();
}}finally{await browser.close()}
fs.writeFileSync('vault/2026-10-30-ssr-per-rep/interactions.json',JSON.stringify(results,null,2));console.log(JSON.stringify(results));
})().catch(e=>{console.error(e.message);process.exitCode=1});
